> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 落地 `/api/v1/agent/*` 命名空间的 P0 基础设施：external → agent 改名一批、Bearer + X-Agent-Id 鉴权、slowapi 限流、统一响应壳与错误码、confirm token 存储、`internal/api-keys` 支持 agent_id、前端"Agent 凭据管理"页。

```python
"""Agent API key（BEL-377 external_api_keys 改名而来，见 Agent API 设计稿 §11）。

``/api/v1/agent/*`` 命名空间的应用层凭据；过渡期同时覆盖
``/api/v1/external/*`` 的存量 key。只存 SHA-256 hash，明文仅在创建时
（``POST /api/v1/internal/api-keys``）返回一次。

House style 对齐 ``alert_rule.py``：``Integer`` 主键、``Mapped`` 注解、
带 server default 的时区感知时间戳。
"""

from __future__ import annotations

from datetime import datetime

from sqlalchemy import Boolean, DateTime, Index, Integer, String, func
from sqlalchemy.orm import Mapped, mapped_column

from app.db.session import Base


class AgentApiKey(Base):
    __tablename__ = "agent_api_keys"
    __table_args__ = (
        Index("idx_aak_key_hash", "key_hash", unique=True),
        Index("idx_aak_revoked_expires", "is_revoked", "expires_at"),
        Index("idx_aak_agent_id", "agent_id"),
    )

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    # SHA-256 hex digest (64 chars) of the plaintext key. Unique — this is
    # what the auth path looks up.
    key_hash: Mapped[str] = mapped_column(String(64), nullable=False)
    # Short non-secret prefix (e.g. ``bbx_ab12cd34``) for display in the
    # admin list so operators can tell keys apart without the secret.
    key_prefix: Mapped[str] = mapped_column(String(24), nullable=False)
    label: Mapped[str] = mapped_column(String(128), nullable=False)
    # Reserved for per-key scope restrictions; P0 不启用校验。
    scope: Mapped[str] = mapped_column(String(64), nullable=False, default="read")
    is_revoked: Mapped[bool] = mapped_column(Boolean, nullable=False, default=False)
    expires_at: Mapped[datetime | None] = mapped_column(
        DateTime(timezone=True), nullable=True
    )
    created_by: Mapped[str] = mapped_column(String(128), nullable=False)
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), nullable=False, server_default=func.now()
    )
    last_used_at: Mapped[datetime | None] = mapped_column(
        DateTime(timezone=True), nullable=True
    )
    # 绑定的 Agent 稳定标识（业务主键，外部 Agent 平台生成）。
    # 存量 external key 为 NULL → 只能读遗留 /external/* 命名空间。
    agent_id: Mapped[str | None] = mapped_column(String(128), nullable=True)
    # 预留授权范围（设计稿 §9 #2），P0 不强制校验。
    project_scope: Mapped[str] = mapped_column(
        String(255), nullable=False, server_default="*"
    )
```

```python
    db: Session,
```

```python
"""Agent API audit-log retention sweep（external_audit_cleanup 改名，设计稿 §11）。

Daily range-delete of ``agent_api_audit_log`` rows older than
``agent_api_audit_retention_days`` (default 90).

队列：default（短任务，每日一次）。
"""

from __future__ import annotations

import logging
from datetime import datetime, timedelta, timezone

from celery import shared_task
from sqlalchemy import delete

logger = logging.getLogger(__name__)


@shared_task(name="agent_api_audit_cleanup", acks_late=False)
def agent_api_audit_cleanup() -> str:
    """Delete agent API audit rows past retention. Idempotent — acks_late=False."""
    from app.core.config import get_settings
    from app.db.session import SessionLocal
    from app.models.agent_api_audit_log import AgentApiAuditLog

    settings = get_settings()
    days = settings.agent_api_audit_retention_days
    cutoff = datetime.now(timezone.utc) - timedelta(days=days)

    db = SessionLocal()
    try:
        result = db.execute(
            delete(AgentApiAuditLog).where(AgentApiAuditLog.created_at < cutoff)
        )
        db.commit()
        deleted = result.rowcount or 0
    finally:
```

```python
@dataclass
class AgentKeyVerdict:
    """agent 通道单候选解析结果（与 external 的 KeyVerdict 区分）。"""

    status: str  # "ok" | "expired" | "revoked" | "not_found" | "agent_mismatch"
    key: AgentApiKey | None = None
    # last_used_at 是否被本次 stamped（caller 决定 commit/rollback）
    changed: bool = False

    @property
    def ok(self) -> bool:
        return self.status == "ok"


LAST_USED_THROTTLE_SECONDS = 60


def resolve_key_for_agent(
    db: Session,
    candidate: str,
    *,
    agent_id: str,
    now: datetime | None = None,
) -> AgentKeyVerdict:
    """agent 通道解析：严格 agent_id 绑定 + last_used_at 节流。

    与 ``resolve_key``（external 通道、多候选、每次 stamp）不同：
    * ``agent_id IS NULL`` 的存量 key 一律 agent_mismatch（设计稿 §2.2 行为矩阵）；
    * 距上次 stamp 不足 60s 跳过更新，避免高频读造成热写。
    """
    now = now or datetime.now(timezone.utc)
    row = db.scalars(
        select(AgentApiKey).where(AgentApiKey.key_hash == hash_key(candidate))
    ).first()
    if row is None:
        return AgentKeyVerdict(status="not_found")
    if row.is_revoked:
        return AgentKeyVerdict(status="revoked", key=row)
    if row.expires_at is not None:
        expires = row.expires_at
        if expires.tzinfo is None:
            expires = expires.replace(tzinfo=timezone.utc)
        if expires <= now:
            return AgentKeyVerdict(status="expired", key=row)
    if row.agent_id is None or row.agent_id != agent_id:
        return AgentKeyVerdict(status="agent_mismatch", key=row)

    changed = False
    last = row.last_used_at
    if last is not None and last.tzinfo is None:
        last = last.replace(tzinfo=timezone.utc)
    if last is None or (now - last).total_seconds() >= LAST_USED_THROTTLE_SECONDS:
        row.last_used_at = now
        changed = True
    return AgentKeyVerdict(status="ok", key=row, changed=changed)
```

```python
from fastapi.exceptions import RequestValidationError

AGENT_PATH_PREFIX = "/api/v1/agent/"


class ErrorCode(str, Enum):

class AgentApiError(Exception):
    """agent 命名空间统一业务异常；handler 序列化为响应壳。"""

    def __init__(
        self,
        code: ErrorCode,
        *,
        reason: str | None = None,
        human_message: str | None = None,
        agent_hint: str | None = None,
        details: list[Any] | dict[str, Any] | None = None,
    ) -> None:
        super().__init__(code.value)
        self.code = code
        self.reason = reason
        self.human_message = human_message
        self.agent_hint = agent_hint
        self.details = details

    @property
    def status_code(self) -> int:
        return _HTTP_STATUS[self.code]


def error_payload(
    code: ErrorCode,
    *,
    reason: str | None = None,
    human_message: str | None = None,
    agent_hint: str | None = None,
    details: list[Any] | dict[str, Any] | None = None,
) -> dict:
    """构造统一失败壳（设计稿 §5.1）。"""
    default_human, default_hint = _DEFAULTS[code]
    error: dict[str, Any] = {
        "code": code.value,
        "human_message": human_message or default_human,
        "agent_hint": agent_hint or default_hint,
    }
    if reason:
        error["reason"] = reason
    if details is not None:
        error["details"] = details
    return {"ok": False, "error": error}


def register_agent_error_handlers(app: FastAPI) -> None:
    """全局异常处理器：agent 前缀统一响应壳；其余路径保持现状。"""

    @app.exception_handler(AgentApiError)
    async def _agent_api_error(_request: Request, exc: AgentApiError) -> JSONResponse:
        return JSONResponse(
            status_code=exc.status_code,
            content=error_payload(
                exc.code,
                reason=exc.reason,
                human_message=exc.human_message,
                agent_hint=exc.agent_hint,
                details=exc.details,
            ),
        )

    @app.exception_handler(RequestValidationError)
    async def _validation_error(
        request: Request, exc: RequestValidationError
    ) -> JSONResponse:
        if request.url.path.startswith(AGENT_PATH_PREFIX):
            details = [
                {
                    "loc": [str(part) for part in err.get("loc", [])],
                    "msg": err.get("msg"),
                    "type": err.get("type"),
                }
                for err in exc.errors()
            ]
            return JSONResponse(
                status_code=400,
                content=error_payload(ErrorCode.VALIDATION_FAILED, details=details),
            )
        # 非 agent 路径：保持 FastAPI 默认行为（422 + detail）
        return JSONResponse(
            status_code=422, content={"detail": jsonable_encoder(exc.errors())}
        )

    @app.exception_handler(StarletteHTTPException)
    async def _http_exception(
        request: Request, exc: StarletteHTTPException
    ) -> JSONResponse:
        if request.url.path.startswith(AGENT_PATH_PREFIX):
            code = (
                ErrorCode.RESOURCE_NOT_FOUND
                if exc.status_code == 404
                else ErrorCode.INTERNAL_ERROR
            )
            return JSONResponse(
                status_code=exc.status_code,
                content=error_payload(
                    code,
                    human_message=str(exc.detail) if exc.detail else None,
                ),
            )
        headers = getattr(exc, "headers", None)
        if headers:
            return JSONResponse(
                status_code=exc.status_code,
                content={"detail": exc.detail},
                headers=headers,
            )
        return JSONResponse(
            status_code=exc.status_code, content={"detail": exc.detail}
        )
```

```python
from __future__ import annotations

from typing import Any


def agent_ok(
    data: Any = None, *, idempotency_key: str | None = None, **extra: Any
) -> dict:
    """成功壳：``{"ok": true, "data": ..., "idempotency_key": ...}``。

    ``extra`` 用于 confirm 预览等附加字段（requires_confirmation 等）。

    if idempotency_key is not None:
        payload["idempotency_key"] = idempotency_key
    payload.update(extra)
    return payload
```

```python
"""Agent 命名空间鉴权依赖（设计稿 §2）。

只认 ``Authorization: Bearer <key_secret>`` + ``X-Agent-Id``；
X-API-Key / ?api_key= / ?skip_auth=true / master key 一律不接受
（设计稿 §2.1 对照表）——agent 是服务端消费者，契约保持最窄。
"""

from __future__ import annotations

import logging
from dataclasses import dataclass

from fastapi import Request

from app.api.agent.errors import AgentApiError, ErrorCode
from app.db.session import SessionLocal
from app.services.agent import key_service

logger = logging.getLogger(__name__)

REASON_TOKEN_INVALID = "token_invalid"
REASON_TOKEN_EXPIRED = "token_expired"
REASON_TOKEN_REVOKED = "token_revoked"
REASON_AGENT_MISMATCH = "agent_id_mismatch"

_STATUS_TO_REASON = {
    "not_found": REASON_TOKEN_INVALID,
    "expired": REASON_TOKEN_EXPIRED,
    "revoked": REASON_TOKEN_REVOKED,
    "agent_mismatch": REASON_AGENT_MISMATCH,
}


@dataclass
class AgentCredential:
    key_id: int
    agent_id: str
    label: str


def _extract_bearer(request: Request) -> str | None:
    authorization = request.headers.get("authorization")
    if not authorization or not authorization.lower().startswith("bearer "):
        return None
    token = authorization[len("bearer ") :].strip()
    return token or None


def require_agent_credential(request: Request) -> AgentCredential:
    """校验 Bearer + X-Agent-Id；通过后注入 request.state。"""
    token = _extract_bearer(request)
    if token is None:
        raise AgentApiError(ErrorCode.AGENT_AUTH_FAILED, reason=REASON_TOKEN_INVALID)
    agent_id = (request.headers.get("x-agent-id") or "").strip()
    if not agent_id:
        raise AgentApiError(
            ErrorCode.AGENT_AUTH_FAILED, reason=REASON_AGENT_MISMATCH
        )

    db = SessionLocal()
    try:
        verdict = key_service.resolve_key_for_agent(db, token, agent_id=agent_id)
        if verdict.ok:
            if verdict.changed:
                db.commit()
            else:
                db.rollback()  # 丢弃节流窗口内的 last_used_at 改动
            request.state.agent_id = agent_id
            request.state.agent_api_key_id = verdict.key.id
            return AgentCredential(
                key_id=verdict.key.id, agent_id=agent_id, label=verdict.key.label
            )
        db.rollback()
        reason = _STATUS_TO_REASON.get(verdict.status, REASON_TOKEN_INVALID)
        raise AgentApiError(ErrorCode.AGENT_AUTH_FAILED, reason=reason)
    except AgentApiError:
        raise
    except Exception:  # noqa: BLE001 — agent 通道 fail closed
        logger.exception("agent API key resolution failed")
        db.rollback()
        raise AgentApiError(
            ErrorCode.AGENT_AUTH_FAILED, reason=REASON_TOKEN_INVALID
        ) from None
    finally:
        db.close()
```

- [ ] **Step 2: 实现 confirm.py**

Create `backend/app/api/agent/confirm.py`:

```python
"""confirm 流 token 存储与装饰器（设计稿 §3）。

危险写操作两步：预览（不落库，HTTP 200 + ok:false + requires_confirmation）
→ 带 X-Confirm-Token 重放（落库）。confirm_token 与请求体 SHA-256 绑定，
防止"预览小改动、确认大改动"。

存储：Redis，key = ``agent:confirm:{token}``；物理 TTL = 2×逻辑 TTL，
以便把"逻辑过期"分类为 CONFIRM_TOKEN_EXPIRED（区别于"从未存在"的 INVALID）。
测试经 ``set_confirm_store`` 注入内存实现。
"""

from __future__ import annotations

import asyncio
import functools
import hashlib
import json
import logging
import secrets
import time
from dataclasses import asdict, dataclass
from typing import Any, Callable, Protocol

from fastapi import Request
from fastapi.responses import JSONResponse

logger = logging.getLogger(__name__)

CONFIRM_KEY_PREFIX = "agent:confirm:"
DEFAULT_CONFIRM_TTL_SECONDS = 300


@dataclass
class ConfirmRecord:
    token: str
    agent_id: str
    action: str
    idempotency_key: str | None
    body_hash: str
    created_at: float  # unix timestamp


class ConfirmStore(Protocol):
    def put(self, record: ConfirmRecord, *, physical_ttl_seconds: int) -> None: ...

    def get(self, token: str) -> ConfirmRecord | None: ...

    def delete(self, token: str) -> None: ...


class RedisConfirmStore:
    """生产实现：``app/services/infra/redis`` 的全局连接池。"""

    def put(self, record: ConfirmRecord, *, physical_ttl_seconds: int) -> None:
        from app.services.infra.redis import get_redis_client

        get_redis_client().setex(
            f"{CONFIRM_KEY_PREFIX}{record.token}",
            physical_ttl_seconds,
            json.dumps(asdict(record)),
        )

    def get(self, token: str) -> ConfirmRecord | None:
        from app.services.infra.redis import get_redis_client

        raw = get_redis_client().get(f"{CONFIRM_KEY_PREFIX}{token}")
        if raw is None:
            return None
        try:
            return ConfirmRecord(**json.loads(raw))
        except (json.JSONDecodeError, TypeError, ValueError):
            logger.warning("confirm token payload corrupted: %s", token)
            return None

    def delete(self, token: str) -> None:
        from app.services.infra.redis import get_redis_client

        get_redis_client().delete(f"{CONFIRM_KEY_PREFIX}{token}")


class InMemoryConfirmStore:
    """测试替身：dict + created_at，逻辑过期由 consume 判定。"""

    def __init__(self) -> None:
        self._rows: dict[str, dict[str, Any]] = {}

    def put(self, record: ConfirmRecord, *, physical_ttl_seconds: int) -> None:
        self._rows[record.token] = {
            "record": record,
            "expires_at": time.time() + physical_ttl_seconds,
        }

    def get(self, token: str) -> ConfirmRecord | None:
        row = self._rows.get(token)
        if row is None or time.time() > row["expires_at"]:
            return None
        return row["record"]

    def delete(self, token: str) -> None:
        self._rows.pop(token, None)


_store: ConfirmStore | None = None


def set_confirm_store(store: ConfirmStore | None) -> None:
    """注入自定义 store；传 None 恢复 Redis 默认。"""
    global _store
    _store = store


def get_confirm_store() -> ConfirmStore:
    global _store
    if _store is None:
        _store = RedisConfirmStore()
    return _store


def compute_body_hash(body: bytes) -> str:
    return hashlib.sha256(body).hexdigest()


def issue_confirm_token(
    *,
    agent_id: str,
    action: str,
    idempotency_key: str | None,
    body: bytes,
    ttl_seconds: int = DEFAULT_CONFIRM_TTL_SECONDS,
) -> tuple[ConfirmRecord, int]:
    token = f"cf_{secrets.token_urlsafe(24)}"
    record = ConfirmRecord(
        token=token,
        agent_id=agent_id,
        action=action,
        idempotency_key=idempotency_key,
        body_hash=compute_body_hash(body),
        created_at=time.time(),
    )
    get_confirm_store().put(record, physical_ttl_seconds=ttl_seconds * 2)
    return record, ttl_seconds


class ConfirmReplayError(Exception):
    """重放校验失败。reason ∈ {"invalid", "expired"}。"""

    def __init__(self, reason: str) -> None:
        super().__init__(reason)
        self.reason = reason


def consume_confirm_token(
    *,
    token: str,
    agent_id: str,
    idempotency_key: str | None,
    body: bytes,
    ttl_seconds: int = DEFAULT_CONFIRM_TTL_SECONDS,
) -> ConfirmRecord:
    """校验并消费（用后即删）。不匹配抛 ConfirmReplayError。"""
    store = get_confirm_store()
    record = store.get(token)
    if record is None:
        raise ConfirmReplayError("invalid")
    if time.time() - record.created_at > ttl_seconds:
        raise ConfirmReplayError("expired")
    if (
        record.agent_id != agent_id
        or record.idempotency_key != idempotency_key
        or record.body_hash != compute_body_hash(body)
    ):
        raise ConfirmReplayError("invalid")
    store.delete(token)
    return record


def require_confirmation(
    *,
    action: str,
    build_preview: Callable[[dict], dict] | None = None,
    validate: Callable[[dict], None] | None = None,
):
    """危险写操作两步确认门（设计稿 §3.2）。

    被装饰 handler 必须是 **async** 且签名只接 ``request: Request``
    （请求体经 ``await request.body()`` 读取，FastAPI 会缓存）。

    * 第一次调用（无 X-Confirm-Token）：``validate(body)`` → 签发 token →
      返回 HTTP 200 预览壳（不进 handler）；
    * 第二次调用（带 X-Confirm-Token）：token 校验（body-hash 绑定保证
      请求体一致）→ 执行 handler。
    ``validate`` 抛 ``AgentApiError`` 时在签发 token 之前失败（spec S6）。
    """

    def decorator(handler):
        if not asyncio.iscoroutinefunction(handler):
            raise TypeError(
                "require_confirmation requires an async handler "
                f"(got {handler!r})"
            )

        @functools.wraps(handler)
        async def wrapper(request: Request, *args, **kwargs):
            decision = await _confirm_gate(
                request, action=action, build_preview=build_preview, validate=validate
            )
            if decision is not None:
                return decision
            return await handler(request, *args, **kwargs)

        return wrapper

    return decorator


async def _confirm_gate(
    request: Request,
    *,
    action: str,
    build_preview: Callable[[dict], dict] | None,
    validate: Callable[[dict], None] | None,
) -> JSONResponse | None:
    """返回 JSONResponse = 预览壳（短路）；返回 None = 放行 handler。"""
    from app.api.agent.errors import AgentApiError, ErrorCode

    body_bytes = await request.body()
    try:
        body = json.loads(body_bytes) if body_bytes else {}
    except json.JSONDecodeError:
        raise AgentApiError(
            ErrorCode.VALIDATION_FAILED, human_message="请求体必须是 JSON 对象"
        )
    if not isinstance(body, dict):
        raise AgentApiError(
            ErrorCode.VALIDATION_FAILED, human_message="请求体必须是 JSON 对象"
        )

    agent_id = getattr(request.state, "agent_id", None) or ""
    idempotency_key = request.headers.get("idempotency-key") or None
    confirm_header = (request.headers.get("x-confirm-token") or "").strip()

    if not confirm_header:
        if validate is not None:
            validate(body)  # 业务校验先于 confirm（spec S6）；抛 AgentApiError 直接短路
        record, ttl = issue_confirm_token(
            agent_id=agent_id,
            action=action,
            idempotency_key=idempotency_key,
            body=body_bytes,
            ttl_seconds=DEFAULT_CONFIRM_TTL_SECONDS,
        )
        preview: dict[str, Any] = {"action": action}
        if build_preview is not None:
            preview.update(build_preview(body))
        return JSONResponse(
            status_code=200,
            content={
                "ok": False,
                "requires_confirmation": True,
                "confirm_token": record.token,
                "preview": preview,
                "expires_in_sec": ttl,
                "human_message": "该操作需要用户确认",
                "agent_hint": (
                    "向用户复述上述摘要；用户确认后用同一 Idempotency-Key + "
                    "X-Confirm-Token 与完全相同的请求体重放同一接口"
                ),
            },
        )

    try:
        consume_confirm_token(
            token=confirm_header,
            agent_id=agent_id,
            idempotency_key=idempotency_key,
            body=body_bytes,
        )
    except ConfirmReplayError as exc:
        code = (
            ErrorCode.CONFIRM_TOKEN_EXPIRED
            if exc.reason == "expired"
            else ErrorCode.CONFIRM_TOKEN_INVALID
        )
        raise AgentApiError(code) from None
    return None
```

```ts
/**
 * Agent API 凭据管理封装（对应后端 /api/v1/internal/api-keys）。
 *
 * 走 /api/backend 代理；X-Is-Admin 由代理按 session 自动注入
 * （lib/backend-auth.ts buildBackendAuthHeaders），客户端不需要也不允许自造。

const API_BASE = '/api/backend'

export type AgentApiKeyItem = {
  id: number
  key_prefix: string
  label: string
  scope: string
  agent_id: string | null
  is_revoked: boolean
  expires_at: string | null
  created_at: string
  last_used_at: string | null
}

export type AgentApiKeyCreateResult = AgentApiKeyItem & {
  plaintext_key: string
}

export async function listAgentApiKeys(): Promise<AgentApiKeyItem[]> {
  const res = await fetch(`${API_BASE}/api/v1/internal/api-keys`, {
    cache: 'no-store',
  })
  if (!res.ok) throw new Error(`凭据列表加载失败：HTTP ${res.status}`)
  return res.json()
}

export async function createAgentApiKey(input: {
  label: string
  agent_id: string
  expires_at: string | null
}): Promise<AgentApiKeyCreateResult> {
  const res = await fetch(`${API_BASE}/api/v1/internal/api-keys`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(input),
  })
  if (!res.ok) throw new Error(`凭据创建失败：HTTP ${res.status}`)
  return res.json()
}

export async function revokeAgentApiKey(id: number): Promise<void> {
  const res = await fetch(`${API_BASE}/api/v1/internal/api-keys/${id}/revoke`, {
    method: 'POST',
  })
  if (!res.ok) throw new Error(`凭据吊销失败：HTTP ${res.status}`)
}
```

```ts
import { test, expect, type Page } from '@playwright/test'

const AUTH_FILE = 'tests/.auth/user.json'
const BASE_URL = process.env.E2E_BASE_URL ?? 'http://localhost:3200'

// 与 discovery.spec.ts 保持一致的登录实现
async function loginAsUser(page: Page): Promise<void> {
  const email = process.env.TEST_USER_EMAIL
  const password = process.env.TEST_USER_PASSWORD
  if (!email || !password) {
    throw new Error(
      'loginAsUser requires TEST_USER_EMAIL and TEST_USER_PASSWORD ' +
        '(set in backend/.env.test — gitignored, never committed).'
    )
  }
  const context = page.context()
  try {
    const fs = await import('node:fs')
    if (fs.existsSync(AUTH_FILE)) {
      const cached = JSON.parse(fs.readFileSync(AUTH_FILE, 'utf-8'))
      if (cached.cookies) await context.addCookies(cached.cookies)
      const session = await context.request.get(`${BASE_URL}/api/auth/session`)
      const data = (await session.json()) as { user?: { id?: string | number } }
      if (data?.user?.id) return
    }
  } catch {
    /* fall through */
  }
  const csrfResp = await context.request.get(`${BASE_URL}/api/auth/csrf`)
  const csrfToken = ((await csrfResp.json()) as { csrfToken?: string }).csrfToken
  if (!csrfToken) throw new Error(`NextAuth CSRF fetch failed: ${csrfResp.status()}`)
  const resp = await context.request.post(
    `${BASE_URL}/api/auth/callback/credentials`,
    {
      form: { csrfToken, email, password, json: 'true' },
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      maxRedirects: 0,
    }
  )
  if (resp.status() !== 200 && resp.status() !== 302) {
    throw new Error(
      `NextAuth login failed: ${resp.status()} ${(await resp.text()).slice(0, 200)}`
    )
  }
  try {
    const fs = await import('node:fs')
    fs.mkdirSync('tests/.auth', { recursive: true })
    fs.writeFileSync(AUTH_FILE, JSON.stringify(await context.storageState()), 'utf-8')
  } catch {
    /* best-effort */
  }
}

const TEST_LABEL = `pw-agent-${Date.now()}`
const TEST_AGENT_ID = 'pw-crawl-agent'

test.describe('Agent 凭据管理页', () => {
```

```markdown
# Agent API (`/api/v1/agent/*`)

面向 LLM Agent 的读写 API，替代原 `/api/v1/external/*`（后者过渡期保留，
见 `backend/docs/external-api.md` 顶部弃用说明）。设计文档：
[`docs/agent-api-design.md`](../../docs/agent-api-design.md)。

## 1. 鉴权（强制）

| Header | 必填 | 说明 |
|---|---|---|
| `Authorization` | ✅ | `Bearer <key_secret>`（管理页创建时的明文 Key，仅展示一次） |
| `X-Agent-Id` | ✅ | 与凭据绑定的 Agent ID 完全一致 |
| `Idempotency-Key` | 写时推荐 | ≤64 字符；缺失会在审计行打 `no_idempotency` 风险标记 |
| `X-Confirm-Token` | 危险写回放时 ✅ | 预览响应返回的 `confirm_token` |

不接受：`X-API-Key` 头、`?api_key=` 查询参数、`?skip_auth=true`、master key。
存量 external key（未绑定 agent_id）调本命名空间一律 401 `agent_id_mismatch`。

凭据管理：前端"系统 → Agent 凭据"页（`/dashboard/system/agent-management`），
或 admin 调 `POST/GET /api/v1/internal/api-keys`。

## 2. 端点（P0）

    GET /api/v1/agent/health    存活探针（返回 agent_id + server_time）

读侧端点（`/external/*` 镜像）与写侧端点（货件域）随 P1/P2 上线，
清单以设计稿 §8 阶段表为准。

## 3. 响应壳

成功：`{ "ok": true, "data": ..., "idempotency_key": "..." }`

失败：`{ "ok": false, "error": { "code", "reason"?, "human_message", "agent_hint", "details"? } }`

错误码（稳定，Agent 可依赖）：`AGENT_AUTH_FAILED`(401) /
`AGENT_RATE_LIMIT_EXCEEDED`(429, 附 Retry-After) / `VALIDATION_FAILED`(400) /
`RESOURCE_NOT_FOUND`(404) / `CONFIRM_TOKEN_INVALID`(400) /
`CONFIRM_TOKEN_EXPIRED`(409) / `IDEMPOTENCY_KEY_CONFLICT`(409) /
`INTERNAL_ERROR`(500)。

## 4. curl 示例

    curl -i http://<host>/api/v1/agent/health \
      -H "Authorization: Bearer bbx_xxx" \
      -H "X-Agent-Id: shipment-assistant-prod"

## 5. 限流

读 300 req/min、写 30 req/min（按 agent_id 计，内存计数、按 worker 独立）。
超限返回 429 壳 + `Retry-After`。
```
